IntSqrt:
    push rbx
    push rcx
    push rdx
    mov  rbx, rax
    cmp  rbx, 1
    jle  .SqrtSmall
    mov  rax, rbx
.SqrtLoop:
    mov  rcx, rax
    mov  rax, rbx
    cqo
    idiv rcx
    add  rax, rcx
    sar  rax, 1
    cmp  rax, rcx
    jl   .SqrtLoop
    mov  rax, rcx
.SqrtDone:
    pop  rdx
    pop  rcx
    pop  rbx
    ret
.SqrtSmall:
    mov  rax, rbx
    jmp  .SqrtDone

IntPow:
    push rcx
    push rdx
    mov  rcx, rbx
    mov  rdx, rax
    mov  rax, 1
    cmp  rcx, 0
    jle  .PowDone
.PowLoop:
    imul rax, rdx
    dec  rcx
    jnz  .PowLoop
.PowDone:
    pop  rdx
    pop  rcx
    ret
